SELECT COUNT(*) 
FROM zavkaf_splits zs
LEFT JOIN xml_content_of_load xcol ON zs.base_uid = xcol.base_uid
WHERE xcol.base_uid IS NULL 
  AND zs.`delete` <> '1';

UPDATE zavkaf_splits zs

* Находить строки где комбинация uid нагрузки и uid лектора не уникальна..

SELECT base_uid, lecturer_uid, COUNT(*) AS c
FROM zavkaf_splits
WHERE `delete` = '0' AND `lecturer_uid` <> ''
GROUP BY base_uid, lecturer_uid
HAVING c > 1
